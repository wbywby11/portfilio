```python
class SelectionService:
    def __init__(self):
        self._registry: Dict[str, type[BaseAssetSelector]] = {}
    
    def register(self, selector_type: str, selector_cls: type[BaseAssetSelector]) -> None:
        """注册选择器类型"""
        if not issubclass(selector_cls, BaseAssetSelector):
            raise TypeError(f"{selector_cls} 必须继承 BaseAssetSelector")
        self._registry[selector_type] = selector_cls
    
    def select_assets(
        self,
        selector_type: str,
        prices: pd.DataFrame,
        n_assets: int,
        **params
    ) -> SelectionResult:
        """执行资产选择"""
        selector_cls = self.get_selector_class(selector_type)
        selector = selector_cls()
        result = selector.select(prices, n_assets, **params)
        return result
```

```python
class StrategyAdapterWithSelection(IStrategy):
    """支持资产选择的策略适配器.
    
    在回测时，如果策略启用了资产选择（如动量选择器），
    会先执行资产选择，然后基于选中的资产计算权重。
    """
    
    def __init__(
            self,
            base_strategy: StrategiesWrapper,
            thscodes: list[str],
            enable_selection: bool,
            selection_info: Dict[str, Any]
    ) -> None:
        """初始化适配器"""
        self._base_strategy = base_strategy
        self._thscodes = thscodes
        self._enable_selection = enable_selection
        self._selection_info = selection_info or {}
        
        # 初始化选择服务
        if enable_selection:
            from app.selection.services.selection_service import SelectionService
            from app.selection.selectors.momentum import MomentumSelector
            
            self._selection_service = SelectionService()
            # 注册动量选择器
            self._selection_service.register("momentum", MomentumSelector)
    
    def generate_target_weights(
            self,
            history_prices: pd.DataFrame,
            as_of: pd.Timestamp,
    ) -> TargetWeightSignal:
        """生成目标权重（支持资产选择）"""
        if self._enable_selection:
            # 执行资产选择
            selector_type = self._selection_info.get("selector_type", "momentum")
            n_selected = self._selection_info.get("n_assets", len(self._thscodes))
            selection_params = self._selection_info.get("selection_params", {})
            
            # 从策略参数中提取选择器参数（如果存在）
            base_params = self._base_strategy._params
            if "min_history_days" in base_params:
                selection_params.setdefault("min_history_days", base_params["min_history_days"])
            if "momentum_windows" in base_params:
                selection_params.setdefault("momentum_windows", base_params["momentum_windows"])
            if "tie_break_window" in base_params:
                selection_params.setdefault("tie_break_window", base_params["tie_break_window"])
            
            # 执行资产选择
            selection_result = self._selection_service.select_assets(
                selector_type=selector_type,
                prices=history_prices,
                n_assets=n_selected,
                **selection_params
            )
            
            selected_assets = selection_result.selected_assets
            
            # 只保留选中资产的历史数据
            selected_history = history_prices[selected_assets] if selected_assets else history_prices
        else:
            # 未启用资产选择，使用所有资产
            selected_assets = self._thscodes
            selected_history = history_prices
        
        # 调用基础策略计算权重（基于选中的资产）
        # 移除选择器相关参数
        strategy_params = {k: v for k, v in self._base_strategy._params.items()
                           if k not in ['enable_momentum_selection', 'selector_type',
                                       'asset_selection_count', 'min_history_days',
                                       'momentum_windows', 'tie_break_window']}
        
        weights_series = self._base_strategy._impl.allocate(selected_history, **strategy_params)
        
        # 转换为字典并确保包含所有候选资产（未选中的为0）
        # ... 后续处理逻辑
```

**关键方法说明**:
- `__init__()`: 初始化适配器，注册动量选择器
- `generate_target_weights()`: 
  - 如果启用资产选择，先执行资产选择
  - 从候选池中筛选TopN资产
  - 基于选中的资产计算权重
  - 返回包含所有候选资产的权重（未选中的为0）

## 四、关键参数说明

### 动量选择器参数（6个）

1. **`enable_momentum_selection`** (bool, Optional):
   - 是否启用动量资产选择
   - 默认: `None` (False)

2. **`selector_type`** (str, Optional):
   - 选择器类型
   - 可选值: `"momentum"`, `"reversal"`, `"hybrid"`
   - 默认: `"momentum"`

3. **`asset_selection_count`** (int, Optional):
   - 选择资产数量
   - 范围: 1-100
   - 默认: `8` 或候选池资产数量

4. **`min_history_days`** (int, Optional):
   - 最小历史数据要求（交易日）
   - 范围: 100-500
   - 默认: `260`

5. **`momentum_windows`** (str, Optional):
   - 动量窗口，用逗号分隔
   - 格式: `"22,65,260"` (对应1月/3月/1年)
   - 默认: `"22,65,260"`

6. **`tie_break_window`** (int, Optional):
   - 并列决胜窗口（交易日）
   - 范围: 5-260
   - 默认: `22`

```
  → BacktestByModelRequest (验证，包含可选的动量选择器参数覆盖)
  → 从MongoDB获取模型配置
  → 合并参数（优先级：请求参数 > 模型配置 > 默认值）
  → 构造BacktestRequest
  → BacktestEngine
  → StrategyAdapterWithSelection (如果启用资产选择)
    → SelectionService.select_assets()
      → MomentumSelector.select()
        → momentum_factors.rank_and_select_assets()
  → 基础策略计算权重（基于选中的资产）
```

## 七、注意事项

1. **参数优先级**:
   - 回测时：前端请求参数 > 数据库模型配置 > 默认值
   - 确保前端可以灵活覆盖模型配置

2. **数据库字段**:
   - 所有动量选择器参数都存储在策略模型文档的顶层
   - 字段名与Pydantic模型字段名一致

3. **向后兼容**:
   - 所有动量选择器参数字段都是 `Optional`，确保向后兼容
   - 未启用资产选择的策略可以正常回测

4. **错误处理**:
   - 资产选择失败时，使用所有资产并记录警告
   - 选中的资产数量少于2个时，给出警告
