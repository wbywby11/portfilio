### 方法1：在 request.js 中添加详细日志

```javascript
// 请求拦截器 - 添加日志
service.interceptors.request.use(
  config => {
    // 添加详细日志
    console.log('📤 发送请求:', {
      url: config.url,
      method: config.method,
      baseURL: config.baseURL,
      fullURL: `${config.baseURL}${config.url}`,
      headers: config.headers,
      data: config.data
    });
    
    const token = store.state.user.token;
    if (token) {
      config.headers['Authorization'] = `Bearer ${token}`;
    }
    return config;
  },
  error => {
    console.error('❌ 请求拦截器错误:', error);
    return Promise.reject(error);
  }
);

// 响应拦截器 - 添加日志
service.interceptors.response.use(
  response => {
    console.log('✅ 收到响应:', {
      status: response.status,
      statusText: response.statusText,
      url: response.config.url,
      data: response.data
    });
    const res = response.data;
    return res;
  },
  error => {
    console.error('❌ 响应错误详情:', {
      message: error.message,
      code: error.code,
      config: error.config ? {
        url: error.config.url,
        method: error.config.method,
        baseURL: error.config.baseURL,
        fullURL: `${error.config.baseURL}${error.config.url}`
      } : null,
      response: error.response ? {
        status: error.response.status,
        statusText: error.response.statusText,
        data: error.response.data
      } : null,
      request: error.request ? '请求已发送但无响应' : null
    });
    
    // 原有的错误处理逻辑...
    if (error.response) {
      const { status, data } = error.response;
      // ... 原有代码
    } else if (error.request) {
      console.error('⚠️ 网络错误：请求已发送，但未收到响应');
      console.error('可能的原因：');
      console.error('1. 后端服务未运行');
      console.error('2. 后端服务运行在不同的端口');
      console.error('3. CORS 跨域问题');
      console.error('4. 防火墙或网络问题');
      Message.error('网络错误，请检查网络连接');
    } else {
      Message.error('请求配置错误');
    }
    
    return Promise.reject(error);
  }
);
```

### 方法2：创建网络测试工具函数

创建 `D:\portfilio\src\utils\networkTest.js`：

```javascript
import axios from 'axios';

/**
 * 测试后端连接
 */
export async function testBackendConnection() {
  const baseURL = 'http://127.0.0.1:8000';
  const testEndpoints = [
    '/api/v1/health',
    '/docs',
    '/api/v1/strategy-models/'
  ];
  
  console.log('🔍 开始测试后端连接...');
  console.log('Base URL:', baseURL);
  
  for (const endpoint of testEndpoints) {
    const url = `${baseURL}${endpoint}`;
    try {
      console.log(`\n测试: ${url}`);
      const response = await axios.get(url, { timeout: 5000 });
      console.log(`✅ 成功: ${response.status} ${response.statusText}`);
    } catch (error) {
      if (error.response) {
        console.error(`⚠️ HTTP错误: ${error.response.status} - ${error.response.statusText}`);
      } else if (error.request) {
        console.error(`❌ 网络错误: 无法连接到 ${url}`);
        console.error('   可能原因: 后端服务未运行或端口不正确');
      } else {
        console.error(`❌ 请求错误: ${error.message}`);
      }
    }
  }
}

/**
 * 测试 CORS 配置
 */
export async function testCORS() {
  const baseURL = 'http://127.0.0.1:8000';
  const url = `${baseURL}/api/v1/strategy-models/`;
  
  console.log('🔍 测试 CORS 配置...');
  
  try {
    const response = await fetch(url, {
      method: 'OPTIONS',
      headers: {
        'Origin': window.location.origin,
        'Access-Control-Request-Method': 'GET'
      }
    });
    
    const corsHeaders = {
      'Access-Control-Allow-Origin': response.headers.get('Access-Control-Allow-Origin'),
      'Access-Control-Allow-Methods': response.headers.get('Access-Control-Allow-Methods'),
      'Access-Control-Allow-Headers': response.headers.get('Access-Control-Allow-Headers')
    };
    
    console.log('CORS 响应头:', corsHeaders);
    
    if (corsHeaders['Access-Control-Allow-Origin']) {
      console.log('✅ CORS 配置正常');
    } else {
      console.warn('⚠️ 未检测到 CORS 响应头');
    }
  } catch (error) {
    console.error('❌ CORS 测试失败:', error);
  }
}

/**
 * 检查后端服务状态
 */
export async function checkBackendStatus() {
  const baseURL = 'http://127.0.0.1:8000';
  
  try {
    // 尝试访问 FastAPI docs
    const response = await fetch(`${baseURL}/docs`, { 
      method: 'HEAD',
      mode: 'no-cors' // 避免 CORS 问题
    });
    
    return {
      status: 'running',
      baseURL: baseURL
    };
  } catch (error) {
    return {
      status: 'not_running',
      baseURL: baseURL,
      error: error.message
    };
  }
}
```

在 Vue 组件中使用：

```javascript

export default {
```

## 五、常见网络错误及解决方案

### 1. Network Error / ERR_CONNECTION_REFUSED

**错误信息**: `Network Error` 或 `ERR_CONNECTION_REFUSED`

**可能原因**:
- 后端服务未运行
- 后端服务运行在不同的端口
- 防火墙阻止连接

**解决方案**:
```bash
# 1. 检查后端服务是否运行
cd D:\portfilio-end
python main.py

# 2. 检查端口占用
netstat -ano | findstr :8000

# 3. 如果端口被占用，修改后端端口或前端 baseURL
```

### 2. CORS 跨域错误

**错误信息**: `Access to fetch at 'http://127.0.0.1:8000/...' from origin 'http://localhost:8080' has been blocked by CORS policy`

**解决方案**:

检查后端 CORS 配置（`D:\portfilio-end\main.py` 或相关配置文件）：

```python
from fastapi.middleware.cors import CORSMiddleware

app.add_middleware(
    CORSMiddleware,
    allow_origins=["http://localhost:8080", "http://127.0.0.1:8080", "http://localhost:3000"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)
```

### 3. 超时错误

**错误信息**: `timeout of 15000ms exceeded`

**解决方案**:
- 增加 timeout 值（当前 15000ms）
- 检查后端响应速度
- 检查网络延迟

```javascript
// 在 request.js 中
timeout: 30000  // 增加到 30 秒
```

### 4. 404 Not Found

**错误信息**: `404 Not Found`

**可能原因**:
- API 路径不正确
- 后端路由未注册

**解决方案**:
- 检查 API 路径是否正确
- 访问 `http://127.0.0.1:8000/docs` 查看所有可用端点

### 5. 500 Internal Server Error

**错误信息**: `500 Internal Server Error`

**可能原因**:
- 后端代码错误
- 数据库连接问题

**解决方案**:
- 查看后端日志
- 检查后端控制台输出

## 六、浏览器 Network 面板调试

### 使用步骤：

1. 打开浏览器开发者工具（F12）
2. 切换到 **Network** 标签
3. 刷新页面或触发请求
4. 查看请求列表：
   - **红色**：请求失败
   - **黄色**：请求警告
   - **绿色**：请求成功

### 关键信息：

- **Status**: HTTP 状态码
- **Type**: 请求类型（xhr, fetch, etc.）
- **Time**: 请求耗时
- **Size**: 响应大小
- **Waterfall**: 请求时间线

### 点击请求查看详情：

- **Headers**: 请求头和响应头
- **Payload**: 请求数据
- **Response**: 响应数据
- **Preview**: 格式化响应

## 七、快速测试脚本

创建 `D:\portfilio\test-network.html`：

```html
<!DOCTYPE html>
<html>
<head>
    <title>网络连接测试</title>
    <style>
        body { font-family: Arial, sans-serif; padding: 20px; }
        .test-item { margin: 10px 0; padding: 10px; border: 1px solid #ddd; }
        .success { background-color: #d4edda; }
        .error { background-color: #f8d7da; }
        button { padding: 10px 20px; margin: 5px; }
    </style>
</head>
<body>
    <h1>后端网络连接测试</h1>
    <div>
        <button onclick="testConnection()">测试连接</button>
        <button onclick="testAPI()">测试 API</button>
        <button onclick="testCORS()">测试 CORS</button>
    </div>
    <div id="results"></div>

    <script>
        const baseURL = 'http://127.0.0.1:8000';
        
        function addResult(message, isSuccess) {
            const div = document.createElement('div');
            div.className = `test-item ${isSuccess ? 'success' : 'error'}`;
            div.textContent = message;
            document.getElementById('results').appendChild(div);
        }
        
        async function testConnection() {
            addResult('🔍 测试基础连接...', true);
            try {
                const response = await fetch(`${baseURL}/docs`, { method: 'HEAD' });
                addResult(`✅ 连接成功: ${response.status}`, true);
            } catch (error) {
                addResult(`❌ 连接失败: ${error.message}`, false);
            }
        }
        
        async function testAPI() {
            addResult('🔍 测试 API 端点...', true);
            try {
                const response = await fetch(`${baseURL}/api/v1/strategy-models/`);
                const data = await response.json();
                addResult(`✅ API 调用成功: ${response.status}`, true);
                console.log('API 响应:', data);
            } catch (error) {
                addResult(`❌ API 调用失败: ${error.message}`, false);
            }
        }
        
        async function testCORS() {
            addResult('🔍 测试 CORS...', true);
            try {
                const response = await fetch(`${baseURL}/api/v1/strategy-models/`, {
                    method: 'OPTIONS'
                });
                const corsHeader = response.headers.get('Access-Control-Allow-Origin');
                if (corsHeader) {
                    addResult(`✅ CORS 配置正常: ${corsHeader}`, true);
                } else {
                    addResult('⚠️ 未检测到 CORS 响应头', false);
                }
            } catch (error) {
                addResult(`❌ CORS 测试失败: ${error.message}`, false);
            }
        }
    </script>
</body>
</html>
```

在浏览器中打开此文件进行测试。
